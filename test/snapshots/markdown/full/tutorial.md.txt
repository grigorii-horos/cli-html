  8 }
  9

 17       />

### Key Concepts

Props Interface ::

  Defines the shape of component props for type safety

FC (FunctionComponent) ::

  TypeScript type for functional components

Event Handlers ::

  Functions passed as props to handle user interactions

## Step 5: Create Add Todo Form

Create src/components/AddTodo.tsx:

  1 import { FC, useState, FormEvent } from 'react';
  2 
  3 interface AddTodoProps {
  4   onAdd: (text: string) => void;
  5 }
  6 
  7 export const AddTodo: FC<AddTodoProps> = ({ onAdd }) => {
  8   const [text, setText] = useState('');
  9 
 10   const handleSubmit = (e: FormEvent) => {
 11     e.preventDefault();
 12     if (text.trim()) {
 13       onAdd(text.trim());
 14       setText('');
 15     }
 16   };
 17 
 18   return (
 19     <form onSubmit={handleSubmit}>
 20       <input
 21         type="text"
 22         value={text}
 23         onChange={(e) => setText(e.target.value)}
 24         placeholder="What needs to be done?"
 25       />
 26       <button type="submit">Add</button>
 27     </form>
 28   );
 29 };

### Form Handling Best Practices

 • [✓] Prevent default form submission
 • [✓] Validate input before processing
 • [✓] Clear input after submission
 • [✓] Trim whitespace from user input

## Step 7: Implement Main App Logic

Update src/App.tsx:

  1 import { useState, useEffect } from 'react';
  2 import { Todo, FilterType } from './types';
  3 import { AddTodo } from './components/AddTodo';
  4 import { TodoList } from './components/TodoList';
  5 import { FilterBar } from './components/FilterBar';
  6 import './App.css';
  7 
  8 function App() {
  9   const [todos, setTodos] = useState<Todo[]>([]);
 10   const [filter, setFilter] = useState<FilterType>('all');
 11 
 12   // Load todos from localStorage on mount
 13   useEffect(() => {
 14     const saved = localStorage.getItem('todos');
 15     if (saved) {
 16       setTodos(JSON.parse(saved));
 17     }
 18   }, []);
 19 
 20   // Save todos to localStorage whenever they change
 21   useEffect(() => {
 22     localStorage.setItem('todos', JSON.stringify(todos));
 23   }, [todos]);
 24 
 25   const addTodo = (text: string) => {
 26     const newTodo: Todo = {
 27       id: crypto.randomUUID(),
 28       text,
 29       completed: false,
 30       createdAt: new Date()
 31     };
 32     setTodos([...todos, newTodo]);
 33   };
 34 
 35   const toggleTodo = (id: string) => {
 36     setTodos(todos.map(todo =>
 37       todo.id === id ? { ...todo, completed: !todo.completed } : todo
 38     ));
 39   };
 40 
 41   const deleteTodo = (id: string) => {
 42     setTodos(todos.filter(todo => todo.id !== id));
 43   };
 44 
 45   const filteredTodos = todos.filter(todo => {
 46     if (filter === 'active') return !todo.completed;
 47     if (filter === 'completed') return todo.completed;
 48     return true;
 49   });
 50 
 51   const activeCount = todos.filter(todo => !todo.completed).length;
 52 
 53   return (
 54     <div className="app">
 55       <h1>Todo App</h1>
 56       <AddTodo onAdd={addTodo} />
 57       <TodoList
 58         todos={filteredTodos}
 59         onToggle={toggleTodo}
 60         onDelete={deleteTodo}
 61       />
 62       <FilterBar
 63         currentFilter={filter}
 64         onFilterChange={setFilter}
 65         activeCount={activeCount}
 66       />
 67     </div>
 68   );
 69 }
 70 
 71 export default App;

### State Management Explained

useState ::

  React hook for adding state to functional components

useEffect ::

  React hook for side effects like data fetching or localStorage

Derived State ::

  Values computed from existing state (like filteredTodos)
